create database DB_keyove_Inventario
USE DB_keyove_Inventario;
GO

CREATE TABLE Personas (
    id_Personas INT IDENTITY(1,1) PRIMARY KEY,
    Nombres NVARCHAR(100) NOT NULL,
    Apellidos NVARCHAR(100) NOT NULL,
    Genero CHAR(1) NOT NULL,
    Correo NVARCHAR(100) NULL,
    Teléfono VARCHAR(9) NULL,
    Fecha_Nac DATE NULL,
    Fecha_Registro DATETIME2 NOT NULL DEFAULT SYSDATETIME(),

    CONSTRAINT CK_Personas_Genero
        CHECK (Genero IN ('F','M')),

    CONSTRAINT CK_Personas_Telefono
        CHECK (
            Teléfono IS NULL
            OR (
                LEN(Teléfono) = 9
                AND Teléfono NOT LIKE '%[^0-9]%'
            )
        )
);
GO

CREATE TABLE Roles (
    id_Rol INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(50) NOT NULL,
    Descripcion NVARCHAR(200) NULL,
    Estado BIT NOT NULL DEFAULT 1,
    Fecha_Registro DATETIME2 NOT NULL DEFAULT SYSDATETIME(),

    CONSTRAINT UQ_Roles_Nombre
        UNIQUE (Nombre)
);
GO

CREATE TABLE Usuarios (
    id_Usuario INT IDENTITY(1,1) PRIMARY KEY,
    id_Personas INT NULL,
    id_Rol INT NOT NULL,
    Nombre_Usuario NVARCHAR(100) NOT NULL,
    Contraseña_Hash NVARCHAR(255) NOT NULL,
    Fecha_Registro DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    Estado BIT NOT NULL DEFAULT 1,

    CONSTRAINT UQ_Usuarios_Nombre_Usuario
        UNIQUE (Nombre_Usuario),

    CONSTRAINT FK_Usuarios_Personas
        FOREIGN KEY (id_Personas)
        REFERENCES Personas(id_Personas),

    CONSTRAINT FK_Usuarios_Roles
        FOREIGN KEY (id_Rol)
        REFERENCES Roles(id_Rol)
);
GO

CREATE TABLE Categorias (
    id_Categoria INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(200) NOT NULL,
    Descripcion NVARCHAR(500) NULL,
    Estado BIT NOT NULL DEFAULT 1,
    Fecha_Registro DATETIME2 NOT NULL DEFAULT SYSDATETIME(),

    CONSTRAINT UQ_Categorias_Nombre
        UNIQUE (Nombre)
);
GO

CREATE TABLE Proveedores (
    id_Proveedor INT IDENTITY(1,1) PRIMARY KEY,
    Ruc VARCHAR(11) NOT NULL,
    Razon_Social NVARCHAR(150) NOT NULL,
    Representante NVARCHAR(100) NULL,
    Telefono VARCHAR(9) NULL,
    Correo NVARCHAR(100) NULL,
    Direccion NVARCHAR(150) NULL,
    Estado BIT NOT NULL DEFAULT 1,
    Fecha_Registro DATETIME2 NOT NULL DEFAULT SYSDATETIME(),

    CONSTRAINT UQ_Proveedores_Ruc
        UNIQUE (Ruc),

    CONSTRAINT CK_Proveedores_Ruc
        CHECK (
            LEN(Ruc) = 11
            AND Ruc NOT LIKE '%[^0-9]%'
        ),

    CONSTRAINT CK_Proveedores_Telefono
        CHECK (
            Telefono IS NULL
            OR (
                LEN(Telefono) = 9
                AND Telefono NOT LIKE '%[^0-9]%'
            )
        )
);
GO

CREATE TABLE Productos (
    id_Producto INT IDENTITY(1,1) PRIMARY KEY,
    id_Categoria INT NOT NULL,
    Codigo NVARCHAR(50) NOT NULL,
    Nombre NVARCHAR(100) NOT NULL,
    Descripcion NVARCHAR(500) NULL,
    Marca NVARCHAR(100) NULL,
    Modelo NVARCHAR(100) NULL,
    Precio_Compra DECIMAL(18,2) NOT NULL DEFAULT 0,
    Precio_Venta DECIMAL(18,2) NOT NULL DEFAULT 0,
    Stock_Actual INT NOT NULL DEFAULT 0,
    Stock_Minimo INT NOT NULL DEFAULT 5,
    Unidad_Medida NVARCHAR(30) NOT NULL DEFAULT 'UNIDAD',
    Estado BIT NOT NULL DEFAULT 1,
    Fecha_Registro DATETIME2 NOT NULL DEFAULT SYSDATETIME(),

    CONSTRAINT UQ_Productos_Codigo
        UNIQUE (Codigo),

    CONSTRAINT FK_Productos_Categorias
        FOREIGN KEY (id_Categoria)
        REFERENCES Categorias(id_Categoria),

    CONSTRAINT CK_Productos_Precio_Compra
        CHECK (Precio_Compra >= 0),

    CONSTRAINT CK_Productos_Precio_Venta
        CHECK (Precio_Venta >= 0),

    CONSTRAINT CK_Productos_Stock_Actual
        CHECK (Stock_Actual >= 0),

    CONSTRAINT CK_Productos_Stock_Minimo
        CHECK (Stock_Minimo >= 0)
);
GO

CREATE TABLE Compras (
    id_Compra INT IDENTITY(1,1) PRIMARY KEY,
    id_Proveedor INT NOT NULL,
    id_Usuario INT NOT NULL,
    Tipo_Comprobante NVARCHAR(30) NOT NULL,
    Numero_Comprobante NVARCHAR(50) NOT NULL,
    Fecha_Compra DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    Sub_Total DECIMAL(18,2) NOT NULL DEFAULT 0,
    Igv DECIMAL(18,2) NOT NULL DEFAULT 0,
    Total DECIMAL(18,2) NOT NULL DEFAULT 0,
    Observacion NVARCHAR(300) NULL,
    Estado NVARCHAR(20) NOT NULL DEFAULT 'REGISTRADA',

    CONSTRAINT FK_Compras_Proveedores
        FOREIGN KEY (id_Proveedor)
        REFERENCES Proveedores(id_Proveedor),

    CONSTRAINT FK_Compras_Usuarios
        FOREIGN KEY (id_Usuario)
        REFERENCES Usuarios(id_Usuario),

    CONSTRAINT CK_Compras_Estado
        CHECK (Estado IN ('REGISTRADA','ANULADA')),

    CONSTRAINT CK_Compras_Montos
        CHECK (
            Sub_Total >= 0
            AND Igv >= 0
            AND Total >= 0
            AND Total = Sub_Total + Igv
        )
);
GO


CREATE UNIQUE INDEX UQ_Compras_Comprobante
ON Compras (
    id_Proveedor,
    Tipo_Comprobante,
    Numero_Comprobante
);
GO

CREATE TABLE Detalle_Compra (
    id_Detalle_Compra INT IDENTITY(1,1) PRIMARY KEY,
    id_Compra INT NOT NULL,
    id_Producto INT NOT NULL,
    Cantidad INT NOT NULL,
    Precio_Compra DECIMAL(18,2) NOT NULL,

    Sub_Total AS (
        CONVERT(DECIMAL(18,2), Cantidad * Precio_Compra)
    ) PERSISTED,

    CONSTRAINT FK_Detalle_Compra_Compras
        FOREIGN KEY (id_Compra)
        REFERENCES Compras(id_Compra),

    CONSTRAINT FK_Detalle_Compra_Productos
        FOREIGN KEY (id_Producto)
        REFERENCES Productos(id_Producto),

    CONSTRAINT CK_Detalle_Compra_Cantidad
        CHECK (Cantidad > 0),

    CONSTRAINT CK_Detalle_Compra_Precio
        CHECK (Precio_Compra >= 0),

    CONSTRAINT UQ_Detalle_Compra_Producto
        UNIQUE (id_Compra, id_Producto)
);
GO


CREATE TABLE Ventas (
    id_Venta INT IDENTITY(1,1) PRIMARY KEY,
    id_Usuario INT NOT NULL,
    Tipo_Comprobante NVARCHAR(30) NOT NULL DEFAULT 'BOLETA',
    Documento_Cliente NVARCHAR(11) NULL,
    Numero_Celular VARCHAR(9) NULL,
    Numero_Comprobante NVARCHAR(50) NOT NULL,
    Fecha_Venta DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    Sub_Total DECIMAL(18,2) NOT NULL DEFAULT 0,
    Igv DECIMAL(18,2) NOT NULL DEFAULT 0,
    Total DECIMAL(18,2) NOT NULL DEFAULT 0,
    Metodo_Pago NVARCHAR(50) NOT NULL DEFAULT 'EFECTIVO',
    Observacion NVARCHAR(300) NULL,
    Estado NVARCHAR(20) NOT NULL DEFAULT 'REGISTRADA',

    CONSTRAINT FK_Ventas_Usuarios
        FOREIGN KEY (id_Usuario)
        REFERENCES Usuarios(id_Usuario),

    CONSTRAINT CK_Ventas_Estado
        CHECK (Estado IN ('REGISTRADA','ANULADA')),

    CONSTRAINT CK_Ventas_Metodo_Pago
        CHECK (
            Metodo_Pago IN (
                'EFECTIVO',
                'YAPE',
                'PLIN',
                'TRANSFERENCIA'
            )
        ),

    CONSTRAINT CK_Ventas_Montos
        CHECK (
            Sub_Total >= 0
            AND Igv >= 0
            AND Total >= 0
            AND Total = Sub_Total + Igv
        ),

    CONSTRAINT CK_Ventas_Celular
        CHECK (
            Numero_Celular IS NULL
            OR (
                LEN(Numero_Celular) = 9
                AND Numero_Celular NOT LIKE '%[^0-9]%'
            )
        ),

    CONSTRAINT CK_Ventas_Documento
        CHECK (
            Documento_Cliente IS NULL
            OR (
                LEN(Documento_Cliente) BETWEEN 8 AND 11
                AND Documento_Cliente NOT LIKE '%[^0-9]%'
            )
        )
);
GO


CREATE UNIQUE INDEX UQ_Ventas_Comprobante
ON Ventas (
    Tipo_Comprobante,
    Numero_Comprobante
);
GO

CREATE TABLE Detalle_Venta (
    id_Detalle_Venta INT IDENTITY(1,1) PRIMARY KEY,
    id_Venta INT NOT NULL,
    id_Producto INT NOT NULL,
    Cantidad INT NOT NULL,
    Precio_Venta DECIMAL(18,2) NOT NULL,
    Descuento DECIMAL(18,2) NOT NULL DEFAULT 0,

    Sub_Total AS (
        CONVERT(
            DECIMAL(18,2),
            (Cantidad * Precio_Venta) - Descuento
        )
    ) PERSISTED,

    CONSTRAINT FK_Detalle_Venta_Ventas
        FOREIGN KEY (id_Venta)
        REFERENCES Ventas(id_Venta),

    CONSTRAINT FK_Detalle_Venta_Productos
        FOREIGN KEY (id_Producto)
        REFERENCES Productos(id_Producto),

    CONSTRAINT CK_Detalle_Venta_Cantidad
        CHECK (Cantidad > 0),

    CONSTRAINT CK_Detalle_Venta_Precio
        CHECK (Precio_Venta >= 0),

    CONSTRAINT CK_Detalle_Venta_Descuento
        CHECK (
            Descuento >= 0
            AND Descuento <= Cantidad * Precio_Venta
        ),

    CONSTRAINT UQ_Detalle_Venta_Producto
        UNIQUE (id_Venta, id_Producto)
);
GO

CREATE TABLE Movimientos_Inventario (
    id_Movimiento INT IDENTITY(1,1) PRIMARY KEY,
    id_Producto INT NOT NULL,
    id_Usuario INT NOT NULL,
    id_Compra INT NULL,
    id_Venta INT NULL,
    Tipo_Movimiento NVARCHAR(30) NOT NULL,
    Cantidad INT NOT NULL,
    Stock_Anterior INT NOT NULL,
    Stock_Nuevo INT NOT NULL,
    Motivo NVARCHAR(500) NULL,
    Fecha_Movimiento DATETIME2 NOT NULL DEFAULT SYSDATETIME(),

    CONSTRAINT FK_Movimientos_Productos
        FOREIGN KEY (id_Producto)
        REFERENCES Productos(id_Producto),

    CONSTRAINT FK_Movimientos_Usuarios
        FOREIGN KEY (id_Usuario)
        REFERENCES Usuarios(id_Usuario),

    CONSTRAINT FK_Movimientos_Compras
        FOREIGN KEY (id_Compra)
        REFERENCES Compras(id_Compra),

    CONSTRAINT FK_Movimientos_Ventas
        FOREIGN KEY (id_Venta)
        REFERENCES Ventas(id_Venta),

    CONSTRAINT CK_Movimientos_Tipo
        CHECK (
            Tipo_Movimiento IN (
                'ENTRADA',
                'SALIDA',
                'AJUSTE_ENTRADA',
                'AJUSTE_SALIDA'
            )
        ),

    CONSTRAINT CK_Movimientos_Cantidad
        CHECK (Cantidad > 0),

    CONSTRAINT CK_Movimientos_Stock
        CHECK (
            Stock_Anterior >= 0
            AND Stock_Nuevo >= 0
        ),

    CONSTRAINT CK_Movimientos_Origen
        CHECK (
            (Tipo_Movimiento = 'ENTRADA'
                AND id_Compra IS NOT NULL
                AND id_Venta IS NULL)

            OR

            (Tipo_Movimiento = 'SALIDA'
                AND id_Venta IS NOT NULL
                AND id_Compra IS NULL)

            OR

            (Tipo_Movimiento = 'AJUSTE_ENTRADA'
                AND id_Compra IS NULL
                AND id_Venta IS NULL)

            OR

            (Tipo_Movimiento = 'AJUSTE_SALIDA'
                AND id_Compra IS NULL
                AND id_Venta IS NULL)
        )
);
GO
